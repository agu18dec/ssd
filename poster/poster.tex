  \begin{alertblock}{TL;DR}
    \vspace{0.6em}
    \centering
    \begin{minipage}{0.92\linewidth}\large\raggedright
      \setlength{\parskip}{0.8em}
      {\bfseries\color{cardinalred}Problem.} Self-distillation puts a document into a model's weights, but
      \on{} training wastes most of its tokens.

      {\bfseries\color{cardinalred}Method.} \ssd{}: the student writes the rollout; the teacher takes over only
      where the two disagree.

      {\bfseries\color{cardinalred}Result.} \on{} accuracy with \textbf{45\% fewer supervised tokens}, and less
      forgetting.
    \end{minipage}
    \vspace{0.8em}
  \end{alertblock}

  \begin{block}{Self-Distillation}
    \begin{itemize}
      \item \textbf{Teacher}: the model with the document in context. \textbf{Student}: the same model without it.
      \item The student is trained to match the teacher token by token.
      \item \off{}: teacher writes the rollout. \on{} (SDFT): student writes it.
    \end{itemize}
  \end{block}

  \begin{block}{Why On-Policy Is Slow}
    \centering
    \includegraphics[width=\textwidth]{figures/signal_panel.pdf}\par
    \vspace{-0.8em}\raggedright
    \begin{itemize}
      \item \off{} learns $\sim$5$\times$ faster but plateaus lower.
      \item Along \on{} rollouts the per-token signal \hl{collapses after a few tokens}.
    \end{itemize}
  \end{block}

  \begin{block}{Inflection Tokens}
    \centering
    \includegraphics[width=\textwidth]{figures/influence_positions.pdf}\par
    \vspace{-0.8em}\raggedright
    \begin{itemize}
      \item Where the document changes the next token ($\delta_t > \tau$), the teacher writes.
      \item \on{} keeps the student's token and drifts into \hl{transient states}.
    \end{itemize}
  \end{block}

  \begin{block}{Setup}
    \begin{itemize}
      \item \textbf{Model}: Qwen3-4B-Instruct as student and (frozen) teacher.
      \item \textbf{Tasks}: Company Memo, Wikipedia, Science Q\&A (closed-book QA).
      \item \textbf{Cost}: supervised tokens, i.e.\ positions the loss is computed on.
    \end{itemize}
  \end{block}

  \begin{block}{Speculative Self-Distillation (SSD)}
    \centering
    \includegraphics[width=0.94\textwidth]{figures/main_figure.pdf}
    \vspace{0.3em}
    \begin{columns}[t]
      \begin{column}{0.48\textwidth}
        \begin{itemize}
          \item Student generates by default.
          \item If $D_{\text{switch}}(\pi_\theta \,\|\, \pi_T) > \tau$, the teacher picks the token.
        \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{itemize}
          \item Every position is trained with the same loss.
          \item $\tau\!\to\!0$ is \off{}, $\tau\!\to\!\infty$ is \on{}.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{block}
