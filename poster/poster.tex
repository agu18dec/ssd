  \noindent\begin{tikzpicture}
    \node[draw=cardinalred, line width=3pt, fill=warmgray, rounded corners=10pt, inner xsep=20pt, inner ysep=26pt,
          text width=\dimexpr\linewidth-46pt\relax, align=left, font=\large] (tldr) {%
      \raggedright\setlength{\parskip}{0.7em}\par\vspace*{-0.5em}%
    {\bfseries\color{cardinalred}Problem.} Self-distillation teaches a model a document by training it to match
    itself with the document in context. \on{} rollouts are accurate but spend most tokens where the teacher adds
    little.

    {\bfseries\color{cardinalred}Method.} \ssd{}: the student writes the rollout; the teacher writes only
    \textbf{inflection tokens}, where the two models disagree by more than $\tau$.

    {\bfseries\color{cardinalred}Result.} \ssd{} matches \on{} accuracy with \textbf{45\% fewer supervised
    tokens} on three tasks and \textbf{forgets less}.
    };
    \node[fill=cardinalred, text=white, rounded corners=8pt, inner xsep=30pt, inner ysep=10pt, font=\LARGE]
      at (tldr.north) {\Raleway\bfseries TL;DR};
  \end{tikzpicture}
  \vspace{0.6cm}

  \begin{block}{Self-Distillation}
    \begin{itemize}
      \item \textbf{Teacher} $\pi_T(\cdot\mid x, C)$: the model with privileged context $C$ (a document).
            \textbf{Student} $\pi_\theta(\cdot\mid x)$: the same model without $C$.
    \end{itemize}
    \vspace{-0.3em}
    \[ \mathcal{L}(\theta)=\mathbb{E}_{y\sim\pi_{\text{gen}}}\Big[\tfrac{1}{|y|}\textstyle\sum_t
       D\big(\pi_T(\cdot\mid x,C,y_{<t})\,\|\,\pi_\theta(\cdot\mid x,y_{<t})\big)\Big] \]
    \vspace{-0.5em}
    \begin{itemize}
      \item Methods differ only in the rollout policy $\pi_{\text{gen}}$; \ssd{} mixes the two per token:
        \begin{itemize}
          \item \off{} ($\pi_{\text{gen}}{=}\pi_T$): strong signal, but trains on states the student never visits.
          \item \on{} ($\pi_{\text{gen}}{=}\pi_\theta$, SDFT): no train--test mismatch, but slow.
        \end{itemize}
    \end{itemize}
  \end{block}

  \begin{block}{Why On-Policy Is Slow}
    \centering
    \includegraphics[width=\textwidth]{figures/signal_panel.pdf}\par
    \vspace{-0.8em}\raggedright
    \begin{itemize}
      \item \hl{Signal} = per-token loss: how much the student can still learn from a token.
      \item \off{} learns $\sim$5$\times$ faster but plateaus lower.
      \item Along \on{} rollouts the per-token signal \hl{collapses after a few tokens}.
    \end{itemize}
  \end{block}

  \begin{block}{Inflection Tokens}
    \centering
    \includegraphics[width=\textwidth]{figures/influence_positions.pdf}\par
    \vspace{-0.8em}\raggedright
    \begin{itemize}
      \item Where the document changes the next token ($\delta_t > \tau$), \ssd{} lets the teacher write.
      \item \on{} keeps the student's token and continues into \hl{transient states}: text the student stops
            producing after the update, so supervision on it is wasted.
    \end{itemize}
  \end{block}

\begin{column}{\widewidth}

  \begin{block}{Speculative Self-Distillation (SSD)}
    \centering
    \includegraphics[width=0.84\textwidth]{figures/main_figure.pdf}
    \vspace{0.3em}
    \begin{columns}[t]
      \begin{column}{0.48\textwidth}
        \begin{itemize}
          \item Student generates by default.
          \item If $D_{\text{switch}}(\pi_\theta \,\|\, \pi_T) > \tau$, the teacher picks the token.
        \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{itemize}
          \item Every position is trained with the same loss.
          \item $\tau\!\to\!0$ is \off{}, $\tau\!\to\!\infty$ is \on{}.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{block}

  \begin{block}{SSD Reaches On-Policy Accuracy with 34--57\% Fewer Tokens}
    \centering
    \includegraphics[width=0.76\textwidth]{figures/figure_1_smoothed.pdf}
  \end{block}

  \begin{columns}[t]

      \begin{block}{One Knob, and a Built-in Curriculum}
        \begin{columns}[t]
          \begin{column}{0.49\textwidth}
            \centering
            \includegraphics[width=\textwidth]{figures/tau_frontier.pdf}
          \end{column}
          \begin{column}{0.49\textwidth}
            \centering
            \includegraphics[width=\textwidth]{figures/teacher_contribution.pdf}
          \end{column}
        \end{columns}
        \begin{itemize}
          \item \textbf{Left}: small $\tau$ is cheap but caps accuracy, like \off{}; large $\tau$ approaches \on{} accuracy
                and cost. $\tau \in [0.2, 0.5]$ (JSD) sits on the \hl{Pareto frontier}.
          \item \textbf{Right}: the teacher writes up to 40\% of tokens early on. As the student learns the document the
                switch fires less, so training becomes \on{} \hl{without a schedule}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Less Forgetting}
        \centering
        \includegraphics[width=0.86\textwidth]{figures/forgetting.pdf}
        \raggedright
        \begin{itemize}
          \item \ssd{}'s rollouts stay close to the student's own distribution.
          \item It learns the new domain \hl{without giving up general ability}.
          \item Best in-domain accuracy with the \hl{smallest drop} on general benchmarks.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
